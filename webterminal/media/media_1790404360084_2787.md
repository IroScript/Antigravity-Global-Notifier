# Principle of Least Privilege: Whitelist-only with NOEXEC
# Target: Azure VM 'FatehAli' (User: azureuser)
# Status: DRAFT - DO NOT DEPLOY WITHOUT EXPLICIT APPROVAL AND BACKUP
#
# CHANGE LOG (fixed from previous draft):
# - FIXED: tag syntax was `NOEXEC: NOPASSWD:` (invalid — space-separated tags
#   before the alias list are not parsed the way the old draft assumed).
#   Correct sudoers syntax chains tags with colons, no spaces, immediately
#   before the command list: NOPASSWD:NOEXEC:
#
# CRITICAL DEPLOYMENT PREREQUISITE:
# 1. The default file '/etc/sudoers.d/90-cloud-init-users' currently grants:
#    'azureuser ALL=(ALL) NOPASSWD:ALL'
#    If 90-cloud-init-users remains active, this restricted policy will be
#    bypassed because sudo evaluates rules additively and the LAST matching
#    rule for a given command wins — but a broad NOPASSWD:ALL earlier in
#    evaluation order can still grant access this file tries to deny.
#    To activate, 90-cloud-init-users MUST be removed or renamed with a
#    lower priority prefix (e.g. 00-) so this file's rules are evaluated after it,
#    or simply deleted since NOPASSWD:ALL is being replaced entirely.
# 2. Safety Rule: Always keep an active Azure Serial Console session open, or
#    take an Azure VM Disk Snapshot before modifying sudoers!
# 3. Always validate syntax before installing:
#      sudo visudo -cf /etc/sudoers.d/99-azureuser-restricted
#    A non-zero exit code means DO NOT proceed — fix the syntax first.
# 4. After installing, open a SECOND ssh session (do not close your current
#    one) and test `sudo -l` as azureuser before logging out of the first.
#    If sudo is broken, your current session's sudo cache may still work
#    long enough to revert via the serial console.
# ==============================================================================

# Defaults specification
Defaults    env_reset
Defaults    mail_badpass
Defaults    secure_path="/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin"

# ------------------------------------------------------------------------------
# 1. COMMAND ALIASES (STRICT WHITELIST ONLY)
# ------------------------------------------------------------------------------

# Package management (apt / apt-get)
# Restricted to update, upgrade, and install.
# NOEXEC below prevents pager/GTFOBins escapes (e.g. less/vi/sh spawned
# from within apt's package-configuration scripts or a pager).
Cmnd_Alias APT_CMNDS = /usr/bin/apt update, \
                       /usr/bin/apt-get update, \
                       /usr/bin/apt upgrade, \
                       /usr/bin/apt-get upgrade, \
                       /usr/bin/apt install *, \
                       /usr/bin/apt-get install *, \
                       /usr/bin/apt autoremove, \
                       /usr/bin/apt-get autoremove

# Service management (systemctl)
# Restricted to restart, start, stop, reload, and status of units.
# Deliberately EXCLUDED: systemctl edit (opens $EDITOR as root — GTFOBins),
# systemctl daemon-reexec, systemctl run (arbitrary command as a transient unit).
Cmnd_Alias SYSTEMCTL_CMNDS = /usr/bin/systemctl start *, \
                             /usr/bin/systemctl stop *, \
                             /usr/bin/systemctl restart *, \
                             /usr/bin/systemctl reload *, \
                             /usr/bin/systemctl status *

# Container / Process management (if required for Alco / ERPNext)
Cmnd_Alias DOCKER_CMNDS = /usr/bin/docker compose *

# ------------------------------------------------------------------------------
# 2. ACCESS SPECIFICATION WITH NOEXEC (CORRECTED SYNTAX)
# ------------------------------------------------------------------------------
# NOEXEC prevents the executed command from spawning further processes via
# execve() (e.g. a shell escape from inside a pager or an apt maintainer
# script). Tags are chained with colons immediately before the command list,
# NOT separated by spaces from the alias — that was the bug in the prior draft.
#
# NOTE: NOEXEC relies on LD_PRELOAD interposition inside sudo and does NOT
# work on statically-linked binaries, and has known limited effect on some
# newer glibc/systemd-related binaries. Treat NOEXEC as defense-in-depth,
# not an absolute guarantee — the primary defense is still that dangerous
# binaries (chattr, rm, dd, bash, python, etc.) are never in this whitelist.

azureuser ALL=(ALL) NOPASSWD:NOEXEC: APT_CMNDS
azureuser ALL=(ALL) NOPASSWD:NOEXEC: SYSTEMCTL_CMNDS
azureuser ALL=(ALL) NOPASSWD:NOEXEC: DOCKER_CMNDS

# ------------------------------------------------------------------------------
# 3. EXPLICIT EXCLUSION SUMMARY (DENY-BY-DEFAULT)
# ------------------------------------------------------------------------------
# In sudoers, any command not explicitly matched above is DENIED by default.
# The following dangerous administrative binaries are strictly EXCLUDED
# (listed here for audit clarity only — they do not need explicit "!" deny
# rules because they were never granted):
# - /usr/bin/chattr, /usr/bin/lsattr (cannot modify or inspect immutability)
# - /bin/rm, /usr/bin/rm, /usr/bin/unlink, /usr/bin/find (cannot delete via root)
# - /bin/dd, /sbin/fdisk, /sbin/mkfs* (cannot overwrite raw block devices)
# - /bin/mount, /bin/umount (cannot remount filesystems read-write/exec)
# - /usr/bin/visudo, /bin/su, /usr/bin/pkexec (cannot modify sudoers or spawn root)
# - /bin/bash, /bin/sh, /usr/bin/python*, /usr/bin/perl (cannot spawn root shells)
# - systemctl edit, apt-config, dpkg-reconfigure (known GTFOBins root-shell vectors)
# ==============================================================================
